# Make sure the server is running on port 5000:
# npm run server

# 1. Check if server is alive
curl http://localhost:5000/api/health

# 2. View all recipes
curl http://localhost:5000/api/recipes

# 3. Search recipes (by name or ingredient)
curl "http://localhost:5000/api/recipes?search=chicken"
curl "http://localhost:5000/api/recipes?ingredient=garlic"

# 4. Create a new recipe
curl -X POST http://localhost:5000/api/recipes \
  -H "Content-Type: application/json" \
  -d '{
    "title": "Quick Garlic Pasta",
    "category": "Dinner",
    "cookingTime": 15,
    "ingredients": [
      { "name": "Pasta", "amount": "200g" },
      { "name": "Garlic", "amount": "3 cloves" }
    ],
    "steps": [
      { "stepNumber": 1, "instruction": "Boil pasta." },
      { "stepNumber": 2, "instruction": "Toss with sautéed garlic." }
    ]
  }'

# 5. Like a recipe (replace ID with an actual recipe _id)
curl -X POST http://localhost:5000/api/recipes/PASTE_ID_HERE/like

# 6. Save a recipe
curl -X POST http://localhost:5000/api/recipes/PASTE_ID_HERE/save

# 7. Delete a recipe
curl -X DELETE http://localhost:5000/api/recipes/PASTE_ID_HERE
